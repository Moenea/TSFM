\begin{algorithm}[H]
\DontPrintSemicolon
\caption{Operator-augmented ensemble of TSFMs for few-shot fault prognosis}
\label{alg:operator-ensemble}
\Input{window stream $\{X_t\}$; target $y$; few-shot ratio $\rho$; pretrained TSFM $\theta_0$}
\Output{union alarm decisions $\{A_t^{U}\}$ and per-operator forecasts on the test split}
\BlankLine
Fit control limits $(\ell,u)$ and scaler on healthy training data \Eqc*[r]{Eq.~(3),~(9)}
Draw nested few-shot subset $\mathcal{S}_{\rho}$ by seeded permutation \Eqc*[r]{Eq.~(8)}
Few-shot align the backbone $\theta$ from $\theta_0$ on $\mathcal{S}_{\rho}$ \Eqc*[r]{Eq.~(7)}
\For{each view $v \in \{$\emph{raw}, \emph{diff}, \emph{diff2}$\}$}{
  Infer $\hat{y}^{(v)}$ on validation and test; restore the difference views \Eqc*[r]{Eq.~(10)--(13)}
}
Map each operator forecast to an alarm decision \Eqc*[r]{Eq.~(14)}
Fuse the diff and diff2 decisions by the union rule \Eqc*[r]{Eq.~(18)}
Score the union decisions for lead time and preFAR \Eqc*[r]{Eq.~(16),~(17)}
\Return union decisions $\{A_t^{U}\}$ and per-operator forecasts\;
\end{algorithm}
